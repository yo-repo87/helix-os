# Raspberry Pi bootloader config — Helix OS
# Compatible with Pi 4 and Pi 5 (aarch64).
# Place this file at the root of the FAT32 USB partition.

# ── ARM64 kernel ──────────────────────────────────────────────────────────────
arm_64bit=1
kernel=boot/vmlinuz
initramfs boot/initrd.img followkernel

# ── GPU memory (minimal — we're a headless NAS OS) ───────────────────────────
gpu_mem=16

# ── UART console for debugging ────────────────────────────────────────────────
enable_uart=1

# ── USB boot (Pi 4: requires OTP fuse set; Pi 5: enabled by default) ─────────
# To set USB boot on Pi 4: sudo rpi-eeprom-config --edit → BOOT_ORDER=0xf14
# Pi 5: USB boot is in BOOT_ORDER by default.

# ── Device tree overlays ──────────────────────────────────────────────────────
# Uncomment if using USB storage with UASP:
# dtoverlay=disable-bt

# ── Pi 5 PCIe (optional — NVMe hat) ──────────────────────────────────────────
# dtparam=pciex1_gen=3
